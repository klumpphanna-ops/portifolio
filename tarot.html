// Arquivo: Controller.java package com.taroapp.controller; import org.springframework.web.bind.annotation.*; @RestController @RequestMapping("/api") public class Controller { // Endpoint para o consulente enviar uma pergunta @PostMapping("/pergunta") public String receberPergunta(@RequestBody String pergunta) { return "Pergunta recebida: " + pergunta + " | Em breve a cartomante responderá."; } // Endpoint para a cartomante responder @PostMapping("/resposta") public String enviarResposta(@RequestBody String resposta) { return "Resposta da cartomante: " + resposta; } // Endpoint para listar clientes (simples, sem banco ainda) @GetMapping("/clientes") public String listarClientes() { return "Lista de clientes: [Cliente1, Cliente2, Cliente3]"; } }
